# The analogous 150/300 MeV/u screening configs reuse existing 256k TOPAS
# replay data and are named beam_minibeam_water_secondary_c12_replay_e{150,300}_*.yaml.

# Controlled C12 energy-loss closure. Regenerate an entry-aligned identity
# replay, then create all primary/secondary scale/straggling/FE cases.
energy_root=/mnt/sda/wuwei/minibeam_secondary_c12_energy_path_e250_20260919
python3 benchmark/carbonminibeam/prepare_water_entry_replay.py \
  /mnt/sda/wuwei/minibeam_water_replay_e250_em12800k/input/water_entrance_primary_c12.phsp \
  --header /mnt/sda/wuwei/minibeam_water_replay_e250_em12800k/input/water_entrance_primary_c12.header \
  --output-dir "$energy_root/input_aligned" \
  --water-entrance-mm 60 --gpu-entry-epsilon-mm 0.0001 \
  --gpu-source-z-mm 0.0001 --selection primary-c12
python3 benchmark/carbonminibeam/prepare_secondary_c12_energy_path_cases.py \
  --output-root "$energy_root"

energy_cases=(
  primary_nostrag_scale1 secondary_nostrag_scale1
  primary_strag_scale1 primary_strag_scale09958
  secondary_strag_scale1 secondary_strag_scale09958
  secondary_legacy_scale09958 secondary_fe010_scale09958
  secondary_fe005_scale09958 secondary_legacy_scale09958_noplanes
  secondary_fe010_scale09958_noplanes secondary_fe005_scale09958_noplanes
)
for case_name in "${energy_cases[@]}"; do
  case_dir="$energy_root/$case_name"
  "$binary" --config "$case_dir/config.yaml" --device cuda \
    --output "$case_dir/depth.csv" --voxel-dose-mhd "$case_dir/dose.mhd" \
    --write-canonical-config "$case_dir/config.canonical.yaml" \
    >"$case_dir/run.log" 2>&1
done

python3 benchmark/carbonminibeam/analyze_secondary_c12_energy_path.py \
  --root "$energy_root" --histories 1622795
