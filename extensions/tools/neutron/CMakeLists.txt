cmake_minimum_required(VERSION 3.22)
project(MaigoNeutronExtract LANGUAGES CXX)
find_package(Geant4 11.3.2 EXACT REQUIRED)
include(${Geant4_USE_FILE})
add_executable(extract_neutron extract.cc)
target_compile_features(extract_neutron PRIVATE cxx_std_17)
target_link_libraries(extract_neutron ${Geant4_LIBRARIES})
add_executable(extract_neutron_recoil extract_recoil.cc)
target_compile_features(extract_neutron_recoil PRIVATE cxx_std_17)
target_link_libraries(extract_neutron_recoil ${Geant4_LIBRARIES})
target_compile_options(extract_neutron_recoil PRIVATE -g)
target_link_options(extract_neutron_recoil PRIVATE -rdynamic)

target_compile_options(extract_neutron PRIVATE -g)
target_link_options(extract_neutron PRIVATE -rdynamic)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(inspect_neutron inspect.cc "${REPO}/src/neutron_package.cpp")
target_include_directories(inspect_neutron PRIVATE "${REPO}/include")
target_compile_features(inspect_neutron PRIVATE cxx_std_20)
