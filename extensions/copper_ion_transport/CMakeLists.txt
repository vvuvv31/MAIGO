cmake_minimum_required(VERSION 3.22)
project(CopperIonTransportExtraction LANGUAGES CXX)
find_package(Geant4 11.3.2 EXACT REQUIRED)
include(${Geant4_USE_FILE})
add_executable(extract extract.cpp)
target_compile_features(extract PRIVATE cxx_std_20)
target_link_libraries(extract PRIVATE ${Geant4_LIBRARIES})
add_executable(validate validate.cpp ../../src/urban_mcs_package.cpp)
target_compile_features(validate PRIVATE cxx_std_20)
target_include_directories(validate PRIVATE ../../include)
target_link_libraries(validate PRIVATE ${Geant4_LIBRARIES})
add_executable(extract_urban ../tools/urban/extract_ct_urban.cc)
target_compile_features(extract_urban PRIVATE cxx_std_17)
target_compile_definitions(extract_urban PRIVATE MAIGO_URBAN_COPPER_ONLY)
target_link_libraries(extract_urban PRIVATE ${Geant4_LIBRARIES})
add_executable(validate_urban ../tools/urban/validate_urban.cpp ../../src/urban_mcs_package.cpp)
target_compile_features(validate_urban PRIVATE cxx_std_20)
target_include_directories(validate_urban PRIVATE ../../include)
