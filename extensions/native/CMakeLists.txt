cmake_minimum_required(VERSION 3.16)
project(maigo_extraction_native LANGUAGES CXX)
find_package(Geant4 11.3.2 EXACT REQUIRED)
include(${Geant4_USE_FILE})
set(KIT_ROOT "${CMAKE_CURRENT_LIST_DIR}/..")
function(native_tool name source)
  add_executable(${name} "${KIT_ROOT}/${source}")
  target_compile_features(${name} PRIVATE cxx_std_17)
  target_link_libraries(${name} PRIVATE ${Geant4_LIBRARIES})
endfunction()
native_tool(material_metadata tools/unified_em/analytic_density_metadata.cpp)
native_tool(extract_ct_urban tools/urban/extract_ct_urban.cc)
native_tool(extract_copper_urban tools/urban/extract_ct_urban.cc)
target_compile_definitions(extract_copper_urban PRIVATE MAIGO_URBAN_COPPER_ONLY)
native_tool(extract_air_urban tools/urban/extract_ct_urban.cc)
target_compile_definitions(extract_air_urban PRIVATE MAIGO_URBAN_AIR_ONLY)
native_tool(extract_air_ionization tools/ionization/extract_air_ionization.cc)
target_compile_definitions(extract_air_ionization PRIVATE MAIGO_URBAN_AIR_ONLY)
add_executable(validate_urban "${KIT_ROOT}/tools/urban/validate_urban.cpp"
               "${KIT_ROOT}/../src/urban_mcs_package.cpp")
target_include_directories(validate_urban PRIVATE "${KIT_ROOT}/../include")
target_compile_features(validate_urban PRIVATE cxx_std_20)
